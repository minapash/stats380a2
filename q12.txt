# Calculate quarterly GDP growth (%) for a sector within a year range.
# Assumes gdp exists and is ordered by quarter.
# Accepts sector names or numeric factor-level positions.
# Relies on the global gdp dataset, limiting reuse with other datasets.
getGrowth <- function(sectorName, startYear, endYear) {

  # Check inputs
  stopifnot(
    length(sectorName) == 1,
    is.character(sectorName) || is.numeric(sectorName),
    !is.na(sectorName),
    length(startYear) == 1, is.numeric(startYear),
    is.finite(startYear), startYear == floor(startYear),
    length(endYear) == 1, is.numeric(endYear),
    is.finite(endYear), endYear == floor(endYear),
    startYear <= endYear
  )

  # Convert numeric level to sector name
  if (is.numeric(sectorName)) {
    sectorLevels <- levels(factor(gdp$sector))
    stopifnot(
      sectorName == floor(sectorName),
      sectorName >= 1,
      sectorName <= length(sectorLevels)
    )
    sectorName <- sectorLevels[sectorName]
  }

  # Extract sector rows
  sectorData <- subset(gdp, sector == sectorName)
  stopifnot(nrow(sectorData) > 0)

  # Calculate quarterly GDP growth
  rate <- (sectorData$amount[-1] /
             sectorData$amount[-nrow(sectorData)] - 1) * 100

  # Select quarters
  quarter <- sectorData$quarter[-1]

  # Combine into df
  growth <- data.frame(quarter = quarter, rate = rate)

  # Filter by year
  years <- as.numeric(substr(growth$quarter, 1, 4))
  growth <- growth[years >= startYear & years <= endYear, ]

  return(growth)
}

The functions stopifnot(), length(), is.numeric(), is.finite(), floor(), factor(), levels(), subset(), nrow(), data.frame(), as.numeric(), is.character(), isna(), and substr() are in package:base, which is loaded by default. getGrowth() is found in .GlobalEnv.

getAnnualGrowth <- function(df, sectorName, startYear, endYear) {

  # Check GDP data format
  if (!isGDPData(df))
    {stop("Invalid GDP data")}

  # Extract sector rows
  if (is.numeric(sectorName)) 
    {sectorName <- levels(df$sector)[sectorName]}
  
  sectorData <- subset(df, sector == sectorName)
  stopifnot(nrow(sectorData) > 0)

   # Calculate rolling annual totals
  n <- nrow(sectorData)
  annualTotal <- rep(NA, n)

  if (n >= 4) {
    for (i in 4:n) {
      annualTotal[i] <- sum(sectorData$amount[(i-3):i])
    }
  }

  # Calculate annual growth rates

    }
  }

    quarter = sectorData$quarter[-1],
    annualrate = annualrate[-1])
  

  # Filter by year

  growth <- growth[years >= startYear & years <= endYear, ]

  return(growth)}

plotGrowths <- function(df, sectorName, startYear, endYear,
                        col = c("skyblue", "red")) {

  # Get growth rates
  growth <- getGrowths(df, sectorName, startYear, endYear)

  # Set up x positions
  n <- nrow(growth)
  x <- 1:n

  # Set up plot
  plot.new()
  plot.window(xlim = c(0.5, n + 0.5), ylim = c(-20, 20))

  # Draw horizontal grid lines
  for (y in seq(-20, 20, by = 5)) {
    abline(h = y, col = "grey80")
  }

  # Draw quarterly growth bars
  rect(x - 0.3, 0, x + 0.3, growth$quarterlyRate,
       col = col[1], border = col[1])

  # Draw annual growth curve
  lines(x, growth$annualRate, col = col[2], lwd = 2)

  # Add axes
  axis(2, at = seq(-20, 20, by = 5), las = 1)
  q1 <- grepl("Q1", growth$quarter)
  axis(1, at = which(q1),
       labels = substr(as.character(growth$quarter[q1]), 1, 4))

  # Set sector title
  if (is.numeric(sectorName)) {
    sectorName <- levels(df$sector)[sectorName]
  }

  # Add labels and title
  title(xlab = "Year", ylab = "GDP growth (%)",
        main = as.character(sectorName))

  # Add legend
  legend("topright",
         legend = c("Quarterly growth", "Annual growth"),
         fill = c(col[1], NA),
         border = c(col[1], NA),
         lty = c(NA, 1),
         col = col,
         lwd = c(NA, 2), bty = "n")
}

plotSectors <- function(
  sectors = c("Total", "Primary industries",
              "Goods producing industries", "Services industries"),
  startYear, endYear, col = c("skyblue", "red"), df = gdp) {

  # Set up multiple plots
  n <- length(sectors)
  ncol <- 3
  nrow <- ceiling(n / ncol)
  oldpar <- par(no.readonly = TRUE)

  par(mfrow = c(nrow, ncol), mar = c(3, 3, 2, 1),

  # Plot each sector
  for (sectorName in sectors) {
    plotGrowths(df, sectorName, startYear, endYear, col = col)
  }

  # Restore plot settings
  par(oldpar)
}
