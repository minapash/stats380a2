# Create function to calculate GDP growth
getGrowth <- function(sectorName) {
  
  # Extract sector rows
  sectorData <- subset(gdp, sector == sectorName)
  
  # Calculate quarterly GDP growth
  rate <- (sectorData$amount[-1] / sectorData$amount[-nrow(sectorData)] - 1) * 100
  
  # Select quarters
  quarter <- sectorData$quarter[-1]
  
  # Combine into df
  growth <- data.frame(quarter = quarter, rate = rate)
  
  return(growth)}
